## figma回复
在微信小程序中用 Canvas 绘制雷达图，并且要有高级的设计感和流畅的弹性动画，我们需要特别注意网格线条的极细化、半透明的叠加层、文字的排版计算以及弹性缓动函数（Easing Function）。

下面为你提供一套完整、开箱即用且高度还原设计原型的雷达图绘制方案。

🎨 核心色彩与视觉参数
为了匹配我们之前确定的“干净、清爽、活泼”的品牌调性，我定义了以下视觉参数：

网格线：#E5E7EB（非常轻微的浅灰，绝不抢戏）
文字-标签：#6B7280（中灰色，保持呼吸感）
文字-分数：#F97316（品牌橙色，与选中态呼应）
数据面积填充：rgba(249, 115, 22, 0.15)（15% 透明度的橙色，像果冻一样透亮）
数据连线边缘：#F97316（纯橙色，2px 粗细）
数据顶点：纯白底色 #FFFFFF + 橙色边框的精美小圆点。
💻 完整的 JavaScript 绘制逻辑
请将以下代码放入你的 Vue 组件方法中，或提取为一个单独的 .js 模块：

/**
 * 绘制雷达图 (支持入场动画)
 * @param {Object} canvas - Canvas 2D 实例 (小程序 node)
 * @param {Object} ctx - Canvas 2D 绘图上下文
 * @param {Array} scores - 分数数组 [80, 60, 90, 75, 85] (0-100)
 * @param {Array} labels - 维度标签数组 ["淀粉肠指数", "加肉程度", "社交油腻度", "性价比", "抗造性"]
 */
export function drawRadar(canvas, ctx, scores, labels) {
  // 1. 获取系统信息进行 rpx -> px 换算
  const sysInfo = uni.getSystemInfoSync();
  const screenWidth = sysInfo.windowWidth;
  const toPx = (rpx) => (rpx / 750) * screenWidth;

  // 2. 初始化画布尺寸
  const canvasWidth = toPx(650);
  const canvasHeight = toPx(560);
  
  // 确保高清屏不模糊（重要：需要在外层处理好 canvas.width = width * dpr）
  const centerX = canvasWidth / 2;
  const centerY = canvasHeight / 2 + toPx(10); // 整体稍微往下挪一点，给顶部文字留空间
  const maxRadius = toPx(170); // 雷达图网格最大半径
  
  const numAxes = 5; // 5个维度
  const angleStep = (Math.PI * 2) / numAxes;
  const startAngle = -Math.PI / 2; // -90度，确保第一个角在正上方

  // 缓动函数 (easeOutBack - 带有果冻回弹感的曲线)
  const easeOutBack = (t) => {
    const c1 = 1.70158;
    const c3 = c1 + 1;
    return 1 + c3 * Math.pow(t - 1, 3) + c1 * Math.pow(t - 1, 2);
  };

  // 动画配置
  const duration = 600; // 动画时长 600ms
  let startTime = null;

  // 核心绘制帧函数
  const drawFrame = (timestamp) => {
    if (!startTime) startTime = timestamp;
    const runtime = timestamp - startTime;
    let progress = Math.min(runtime / duration, 1);
    
    // 应用弹性缓动曲线
    const easedProgress = easeOutBack(progress);

    // 清空画布
    ctx.clearRect(0, 0, canvasWidth, canvasHeight);

    // === 第一步：绘制背景网格 ===
    ctx.strokeStyle = '#E5E7EB'; // 浅灰网格线
    ctx.lineWidth = toPx(2); // 极细线
    
    const numLevels = 4; // 画4圈五边形
    for (let level = 1; level <= numLevels; level++) {
      const r = maxRadius * (level / numLevels);
      ctx.beginPath();
      for (let i = 0; i < numAxes; i++) {
        const angle = startAngle + i * angleStep;
        const x = centerX + Math.cos(angle) * r;
        const y = centerY + Math.sin(angle) * r;
        if (i === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
      }
      ctx.closePath();
      ctx.stroke();
    }

    // 绘制放射状轴线
    for (let i = 0; i < numAxes; i++) {
      const angle = startAngle + i * angleStep;
      ctx.beginPath();
      ctx.moveTo(centerX, centerY);
      ctx.lineTo(
        centerX + Math.cos(angle) * maxRadius,
        centerY + Math.sin(angle) * maxRadius
      );
      ctx.stroke();
    }

    // === 第二步：绘制角落的文字与分数 ===
    // 提示：文字不需要动画，直接绘制
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    
    for (let i = 0; i < numAxes; i++) {
      const angle = startAngle + i * angleStep;
      // 文字的距离要比网格大一点，预留出 Padding
      const textR = maxRadius + toPx(45); 
      let x = centerX + Math.cos(angle) * textR;
      let y = centerY + Math.sin(angle) * textR;

      // 微调文字位置防止重叠
      if (angle === startAngle) y -= toPx(10); // 顶部的字再往上推一点

      // 绘制标签 (灰色)
      ctx.fillStyle = '#6B7280';
      ctx.font = `${toPx(24)}px sans-serif`;
      ctx.fillText(labels[i], x, y - toPx(16));

      // 绘制分数 (品牌橙)
      ctx.fillStyle = '#F97316';
      ctx.font = `bold ${toPx(32)}px sans-serif`;
      ctx.fillText(scores[i], x, y + toPx(16));
    }

    // === 第三步：绘制数据图层 (带扩展动画) ===
    ctx.beginPath();
    for (let i = 0; i < numAxes; i++) {
      const angle = startAngle + i * angleStep;
      // 计算当前半径 = 最大半径 * (分数/100) * 动画进度
      const valRadius = maxRadius * (scores[i] / 100) * easedProgress;
      const x = centerX + Math.cos(angle) * valRadius;
      const y = centerY + Math.sin(angle) * valRadius;
      
      if (i === 0) ctx.moveTo(x, y);
      else ctx.lineTo(x, y);
    }
    ctx.closePath();

    // 填充半透明橙色
    ctx.fillStyle = 'rgba(249, 115, 22, 0.15)';
    ctx.fill();

    // 绘制数据实线边框
    ctx.strokeStyle = '#F97316';
    ctx.lineJoin = 'round'; // 让转角圆润
    ctx.lineWidth = toPx(4); // 2px 粗细
    ctx.stroke();

    // 绘制数据顶点的小圆圈
    for (let i = 0; i < numAxes; i++) {
      const angle = startAngle + i * angleStep;
      const valRadius = maxRadius * (scores[i] / 100) * easedProgress;
      const x = centerX + Math.cos(angle) * valRadius;
      const y = centerY + Math.sin(angle) * valRadius;

      ctx.beginPath();
      ctx.arc(x, y, toPx(6), 0, Math.PI * 2);
      ctx.fillStyle = '#FFFFFF'; // 白底
      ctx.fill();
      ctx.lineWidth = toPx(4);
      ctx.strokeStyle = '#F97316'; // 橙框
      ctx.stroke();
    }

    // 如果动画还没结束，请求下一帧
    if (progress < 1) {
      canvas.requestAnimationFrame(drawFrame);
    }
  };

  // 启动动画
  canvas.requestAnimationFrame(drawFrame);
}
📦 在 Vue 2 / uni-app 中的调用方式（微信小程序 Canvas 2D）
小程序新版 Canvas 2D 接口性能极佳，非常适合做动画。你的 Vue 模板和生命周期应该这样配合：

模板 template：

<view class="radar-container">
  <!-- 必须指定 type="2d"，宽高写死 rpx 然后在 css 里处理 -->
  <canvas type="2d" id="radarCanvas" class="radar-canvas"></canvas>
</view>
样式 css：

.radar-container {
  display: flex;
  justify-content: center;
  align-items: center;
}
.radar-canvas {
  width: 650rpx;
  height: 560rpx;
}
逻辑 script：

export default {
  data() {
    return {
      scores: [85, 90, 40, 75, 80],
      labels: ["淀粉肠指数", "加肉程度", "社交油腻度", "性价比", "抗造性"]
    }
  },
  mounted() {
    // 延迟一点等 DOM 渲染完再画，或者放在 $nextTick 里
    this.$nextTick(() => {
      this.initRadar();
    });
  },
  methods: {
    initRadar() {
      const query = uni.createSelectorQuery().in(this);
      query.select('#radarCanvas')
        .fields({ node: true, size: true })
        .exec((res) => {
          if (!res[0] || !res[0].node) return;
          
          const canvas = res[0].node;
          const ctx = canvas.getContext('2d');
          
          // 【非常关键】处理设备像素比，防止高清屏变糊！
          const dpr = uni.getSystemInfoSync().pixelRatio;
          canvas.width = res[0].width * dpr;
          canvas.height = res[0].height * dpr;
          ctx.scale(dpr, dpr);

          // 调用我们写好的绘制函数
          drawRadar(canvas, ctx, this.scores, this.labels);
        });
    }
  }
}
✨ 这个方案的优势：
自带物理弹性动画：引入了 easeOutBack 贝塞尔算法，雷达图长出来时，会“冲出去一点点再弹回来”，完全契合你全局的“果冻按压”和“弹性掉落”心智。
绝对防糊：通过 sysInfo.pixelRatio 获取 DPR 并对 Canvas 进行高清缩放，文字和线条在最新的 iPhone Pro 上依然锋利清晰。
完美响应 rpx：内置了 toPx() 换算，无论在什么机型上，比例永远和 650x560 rpx 的设计稿保持绝对一致。

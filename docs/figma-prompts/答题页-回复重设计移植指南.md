页面采用 A/B 双槽位叠加切换。同一时间只有 activeSlot 可点击，另一槽用于离场动画播放。

7.2 pick 方法时序（伪代码）
pick(slot, i, e) {
  if (this.locked || slot !== this.activeSlot) return
  this.locked = true
  const cur = this.slots[slot]
  cur.sel = i
  cur.confirm = i
  uni.vibrateShort({ type: isLast ? 'heavy' : 'light' })

  // 副作用：ripple / flash / streak / confetti(若最后一题)
  this.spawnRipple(slot, i, e)
  this.flashVisible = true; setTimeout(() => this.flashVisible = false, 220)
  this.checkStreak()
  if (isLast) this.confettiVisible = true

  // 420ms 后撤 confirm 类
  setTimeout(() => { cur.confirm = -1 }, 420)

  // 420ms 选中态保持
  setTimeout(() => {
    if (isLast) { uni.navigateTo({ url:'/result' }); return }
    this.idx++
    cur.phase = 'out'

    // 100ms 后新槽位入场
    const next = slot === 'A' ? 'B' : 'A'
    setTimeout(() => {
      const ns = this.slots[next]
      ns.q = this.questions[this.idx]
      ns.sel = -1
      ns.confirm = -1
      ns.phase = 'in'
      ns.visible = true
      this.activeSlot = next

      // 旧槽离场完成（220ms 后）销毁
      setTimeout(() => {
        cur.visible = false
        this.locked = false
      }, 220)
    }, 100)
  }, 420)
}
